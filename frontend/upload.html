const uploadInput = document.getElementById('uploadFile'); const uploadButton = document.getElementById('uploadBtn'); const resultBox = document.getElementById('resultBox'); const filterSelect = document.getElementById('filterType'); const issueList = document.getElementById('issueList'); let currentIssues = []; uploadButton.addEventListener('click', async () => { const file = uploadInput.files[0]; if (!file) { resultBox.textContent = '请选择要上传的论文文件。'; return; } resultBox.textContent = '正在检查...'; try { const formData = new FormData(); formData.append('file', file); const response = await fetch('http://localhost:8000/api/upload', { method: 'POST', body: formData, }); const data = await response.json(); const summary = data.summary || { error: 0, warning: 0, info: 0 }; resultBox.innerHTML = `

标题: ${data.document_info?.title || '未检测到'}

错误: ${summary.error}   警告: ${summary.warning}   信息: ${summary.info}

`; currentIssues = data.issues || []; renderIssues(currentIssues, filterSelect.value); } catch (error) { resultBox.textContent = '上传失败，请确认后端已启动。'; } }); function renderIssues(issues, filter) { issueList.innerHTML = ''; const visibleIssues = filter === 'all' ? issues : issues.filter(item => item.severity === filter); if (!visibleIssues.length) { issueList.innerHTML = '
  • 没有符合当前筛选条件的问题。
    • '; return; } visibleIssues.forEach((issue) => { const li = document.createElement('li'); li.className = `issue-item ${issue.severity}`; li.innerHTML = `
    ${issue.severity} ${issue.rule_id} ${issue.location || 'general'}
    ${issue.title}

    ${issue.message}

    建议：${issue.suggestion || '请参考模板要求修正。'}

    批注编号：${issue.annotation_id || 'N/A'}

    `; issueList.appendChild(li); }); } filterSelect.addEventListener('change', () => { renderIssues(currentIssues, filterSelect.value); });
